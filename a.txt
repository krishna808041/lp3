import java.util.*;

public class Daa_01_fibo {
    static int fibRecursive(int n) {
        if (n <= 1)
            return n;
        return fibRecursive(n - 1) + fibRecursive(n - 2);
    }

    static void fibNonRecursive(int n) {
        int n1 = 0, n2 = 1;
        System.out.print(n1 + " " + n2 + " ");
        for (int i = 2; i < n; i++) {
            int n3 = n1 + n2;
            System.out.print(n3 + " ");
            n1 = n2;
            n2 = n3;
        }
        System.out.println();
    }

    public static void main(String[] args) {
        // Example run with n = 5
        int n = 5;
        System.out.println("Enter the number of elements: " + n);

        System.out.print("\nFibonacci Sequence (Recursive): ");
        long start1 = System.nanoTime();
        for (int i = 0; i < n; i++) {
            System.out.print(fibRecursive(i) + " ");
        }
        long end1 = System.nanoTime();
        double timeRecursive = (end1 - start1) / 1000.0;

        System.out.print("\n\nFibonacci Sequence (Non-Recursive): ");
        long start2 = System.nanoTime();
        fibNonRecursive(n);
        long end2 = System.nanoTime();
        double timeNonRecursive = (end2 - start2) / 1000.0;

        System.out.println("\n=== Time and Space Complexity Analysis ===");
        System.out.printf("Recursive Time Taken: %.2f microseconds\n", timeRecursive);
        System.out.println("Recursive Time Complexity: O(2^n)");
        System.out.println("Recursive Space Complexity: O(n)");
        
        System.out.printf("\nNon-Recursive Time Taken: %.2f microseconds\n", timeNonRecursive);
        System.out.println("Non-Recursive Time Complexity: O(n)");
        System.out.println("Non-Recursive Space Complexity: O(1)");
    }
}

import java.util.*;

class Node {
    char ch;
    int freq;
    Node left, right;
    
    Node(char ch, int freq) {
        this.ch = ch;
        this.freq = freq;
    }
    
    Node(int freq, Node left, Node right) {
        this.freq = freq;
        this.left = left;
        this.right = right;
    }
}

public class Daa_02_huff {
    static void printCodes(Node root, String code) {
        if (root.left == null && root.right == null) {
            System.out.println(root.ch + " -> " + code);
            return;
        }
        if (root.left != null)
            printCodes(root.left, code + "0");
        if (root.right != null)
            printCodes(root.right, code + "1");
    }

    public static void main(String[] args) {
        // Example run with predefined characters and frequencies
        int n = 3;
        char[] chars = {'A', 'B', 'C'};
        int[] freqs = {5, 7, 10};

        System.out.println("Enter number of characters: " + n);

        PriorityQueue<Node> pq = new PriorityQueue<>((a, b) -> a.freq - b.freq);
        
        for (int i = 0; i < n; i++) {
            System.out.println("Enter character: " + chars[i]);
            System.out.println("Enter frequency: " + freqs[i]);
            pq.add(new Node(chars[i], freqs[i]));
        }

        while (pq.size() > 1) {
            Node left = pq.poll();
            Node right = pq.poll();
            Node newNode = new Node(left.freq + right.freq, left, right);
            pq.add(newNode);
        }

        Node root = pq.poll();
        System.out.println("\nHuffman Codes:");
        System.out.println("----------------");
        printCodes(root, "");
    }
}

class Item {
    int weight;
    int value;

    Item(int weight, int value) {
        this.weight = weight;
        this.value = value;
    }
}

public class FractionalKnapsack {

    public static double getMaxValue(Item[] arr, int capacity) {

        // Step 1: Sort by value/weight ratio (descending)
        for (int i = 0; i < arr.length - 1; i++) {
            for (int j = 0; j < arr.length - i - 1; j++) {

                double r1 = (double) arr[j].value / arr[j].weight;
                double r2 = (double) arr[j + 1].value / arr[j + 1].weight;

                if (r1 < r2) {
                    Item temp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = temp;
                }
            }
        }

        // Step 2: Calculate maximum value
        double totalValue = 0.0;
        int currentCapacity = capacity;

        for (Item item : arr) {

            // Take complete item
            if (currentCapacity >= item.weight) {
                currentCapacity -= item.weight;
                totalValue += item.value;
            }
            // Take fractional item
            else {
                double fraction = (double) currentCapacity / item.weight;
                totalValue += item.value * fraction;
                break;
            }
        }

        return totalValue;
    }

    public static void main(String[] args) {

        Item[] arr = {
            new Item(10, 60),
            new Item(20, 100),
            new Item(30, 120)
        };

        int capacity = 50;

        double maxValue = getMaxValue(arr, capacity);

        System.out.println("Maximum value we can obtain = " + maxValue);
    }
}

public class ZeroOneKnapsack {

    public static int knapsack(int W, int wt[], int val[], int n) {

        int K[][] = new int[n + 1][W + 1];

        // Build DP table
        for (int i = 0; i <= n; i++) {
            for (int w = 0; w <= W; w++) {

                if (i == 0 || w == 0) {
                    K[i][w] = 0;
                }
                else if (wt[i - 1] <= w) {
                    K[i][w] = Math.max(
                        val[i - 1] + K[i - 1][w - wt[i - 1]],
                        K[i - 1][w]
                    );
                }
                else {
                    K[i][w] = K[i - 1][w];
                }
            }
        }

        // Print selected items
        int w = W;

        System.out.println("Selected Items:");

        for (int i = n; i > 0 && w > 0; i--) {

            if (K[i][w] != K[i - 1][w]) {

                System.out.println("Item " + i +
                    " -> Weight: " + wt[i - 1] +
                    ", Value: " + val[i - 1]);

                w = w - wt[i - 1];
            }
        }

        return K[n][W];
    }

    public static void main(String[] args) {

        int val[] = {60, 100, 120};
        int wt[] = {10, 20, 30};

        int W = 50;
        int n = val.length;

        System.out.println("Maximum value = " +
            knapsack(W, wt, val, n));
    }
}

import java.util.Scanner;

public class d_n_queens {
    static int count = 0;

    public static boolean isSafe(char board[][], int row, int col) {
        // vertical up direction
        for (int i = row - 1; i >= 0; i--) {
            if (board[i][col] == 'Q') return false;
        }

        // vertical down direction (needed when first queen is placed below current row)
        for (int i = row + 1; i < board.length; i++) {
            if (board[i][col] == 'Q') return false;
        }

        // Diagonally left-up Direction
        for (int i = row - 1, j = col - 1; i >= 0 && j >= 0; i--, j--) {
            if (board[i][j] == 'Q') return false;
        }

        // Diagonally right-up Direction
        for (int i = row - 1, j = col + 1; i >= 0 && j < board[0].length; i--, j++) {
            if (board[i][j] == 'Q') return false;
        }

        // Diagonally left-down Direction (needed when first queen is placed below current row)
        for (int i = row + 1, j = col - 1; i < board.length && j >= 0; i++, j--) {
            if (board[i][j] == 'Q') return false;
        }

        // Diagonally right-down Direction (needed when first queen is placed below current row)
        for (int i = row + 1, j = col + 1; i < board.length && j < board[0].length; i++, j++) {
            if (board[i][j] == 'Q') return false;
        }

        return true;
    }

    public static void nQueens(char board[][], int row, int firstRow) {
        // base case
        if (row == board.length) {
            printBoard(board);
            return;
        }

        // Skip the row where the first queen is already placed
        if (row == firstRow) {
            nQueens(board, row + 1, firstRow);
            return;
        }

        // kaam
        for (int j = 0; j < board.length; j++) {
            if (isSafe(board, row, j)) {
                board[row][j] = 'Q';
                nQueens(board, row + 1, firstRow);
                board[row][j] = '.'; // backtracking
            }
        }
    }

    public static void printBoard(char board[][]) {
        count++;
        System.out.println("-----------Chess Board " + count + "--------------");
        for (int i = 0; i < board.length; i++) {
            for (int j = 0; j < board.length; j++) {
                System.out.print(board[i][j] + " ");
            }
            System.out.println();
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter N (size of board): ");
        int n = sc.nextInt();

        System.out.print("Enter first queen row (0 to " + (n - 1) + "): ");
        int firstRow = sc.nextInt();

        System.out.print("Enter first queen col (0 to " + (n - 1) + "): ");
        int firstCol = sc.nextInt();

        char board[][] = new char[n][n];

        // initialize
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                board[i][j] = '.';
            }
        }

        // Place the first queen at user-specified position
        board[firstRow][firstCol] = 'Q';

        nQueens(board, 0, firstRow);

        if (count == 0) {
            System.out.println("No solution exists");
        }

        System.out.println("Time Complexity: O(N!)");
        System.out.println("Space Complexity: O(N^2)");

        sc.close();
    }
}

# Function definition: y = (x + 3)^2
def objective_function(x):
    return (x + 3) ** 2

# Derivative of the function: dy/dx = 2 * (x + 3)
def derivative_function(x):
    return 2 * (x + 3)

# Initialize parameters
x = 2.0          # Starting point
learning_rate = 0.1  # Step size
epochs = 20      # Number of iterations

print(f"Starting Gradient Descent from x = {x}\n")

# Gradient Descent Loop
for i in range(1, epochs + 1):
    grad = derivative_function(x)
    x = x - learning_rate * grad
    y = objective_function(x)
    
    print(f"Iteration {i:2d}: x = {x:.6f}, y = {y:.6f}")

print(f"\nLocal Minimum found at x = {x:.6f} with value y = {objective_function(x):.6f}")

print("Columns available:", df.columns.tolist()[:8])

# ---------------------------------------------------------
# 2. Pre-process and Select Numerical Features
# ---------------------------------------------------------
# Select relevant numerical columns for clustering (e.g., Quantity, Sales, MSRP)
features = ['QUANTITYORDERED', 'PRICEEACH', 'SALES', 'MSRP']
X = df[features].dropna()

# ---------------------------------------------------------
# 3. Normalize the Data (Standardization)
# ---------------------------------------------------------
# K-Means relies heavily on distance metrics, so scaling is crucial
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# ---------------------------------------------------------
# 4. Determine the Number of Clusters using the Elbow Method
# ---------------------------------------------------------
inertia = []
K_range = range(1, 11)

for k in K_range:
    kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
    kmeans.fit(X_scaled)
    inertia.append(kmeans.inertia_)

# Plot the Elbow Graph
plt.figure(figsize=(8, 4))
plt.plot(K_range, inertia, marker='o', linestyle='-', color='b')
plt.title("The Elbow Method (Optimal K Determination)")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("Inertia (Sum of Squared Distances)")
plt.grid(True)
plt.show()

# ---------------------------------------------------------
# 5. Implement K-Means with Optimal K (e.g., K = 3 or 4)
# ---------------------------------------------------------
optimal_k = 3  # Update based on the visual elbow point from your graph
kmeans_final = KMeans(n_clusters=optimal_k, random_state=42, n_init=10)
clusters = kmeans_final.fit_predict(X_scaled)

# Add cluster labels back to the dataframe subset

print(f"\nSuccessfully clustered data into {optimal_k} groups!")
print(X['Cluster'].value_counts())

contract SimpleBank {
    // Mapping to store the Ether balance of each customer's address
    mapping(address => uint) private balances;
    
    // Address of the bank owner/deployer
    address public owner;

    // Events for logging transactions
    event DepositMade(address indexed accountAddress, uint amount);
    event WithdrawalMade(address indexed accountAddress, uint amount);

    // Constructor to set the contract deployer as the owner
    constructor() {
        owner = msg.sender;
    }

    // 1. Deposit Money: Send Ether into the bank contract
    function deposit() public payable returns (uint) {
        require(msg.value > 0, "Deposit amount must be greater than zero");
        balances[msg.sender] += msg.value;
        emit DepositMade(msg.sender, msg.value);
        return balances[msg.sender];
    }

    // 2. Withdraw Money: Withdraw specified amount of Ether securely using .call()
    function withdraw(uint withdrawAmount) public returns (uint) {
        require(balances[msg.sender] >= withdrawAmount, "Insufficient balance");
        
        balances[msg.sender] -= withdrawAmount;
        
        // Secure modern transfer method
        (bool success, ) = payable(msg.sender).call{value: withdrawAmount}("");
        require(success, "Transfer failed.");

        emit WithdrawalMade(msg.sender, withdrawAmount);
        return balances[msg.sender];
    }

    // 3. Show Balance: Check current Ether balance of the caller
    function getBalance() public view returns (uint) {
        return balances[msg.sender];
    }
}

contract StudentData {
    // 1. Structure to define a Student's attributes
    struct Student {
        uint256 rollNo;
        string name;
        string department;
    }

    // 2. Dynamic Array to store multiple Student records
    Student[] public students;

    // Event to log fallback triggers
    event FallbackCalled(address sender, uint256 value, bytes data);

    // Function to add a new student to the array
    function addStudent(uint256 _rollNo, string memory _name, string memory _department) public {
        students.push(Student(_rollNo, _name, _department));
    }

    // Function to retrieve a specific student by index
    function getStudent(uint256 _index) public view returns (uint256, string memory, string memory) {
        require(_index < students.length, "Student index out of bounds");
        Student memory s = students[_index];
        return (s.rollNo, s.name, s.department);
    }

    // Function to get total number of students stored
    function getTotalStudents() public view returns (uint256) {
        return students.length;
    }

    // 3. Fallback function (triggered when non-existent function is called or Ether is sent with data)
    fallback() external payable {
        emit FallbackCalled(msg.sender, msg.value, msg.data);
    }

    // Receive function (optional companion to fallback for plain Ether transfers)
    receive() external payable {
        emit FallbackCalled(msg.sender, msg.value, "");
    }
}
